1.How do delivery delays impact order profitability?

SELECT
    Delay,
    COUNT(*) AS total_orders,
    ROUND(AVG("Order Profit Per Order"), 2) AS avg_profit,
    ROUND(SUM("Order Profit Per Order"), 2) AS total_profit
FROM supply_chain
GROUP BY Delay
ORDER BY Delay;

2. Which shipping modes have highest delay rate?
SELECT
    "Shipping Mode",
    COUNT(*) AS total_orders,
    SUM(CASE WHEN Delay > 0 THEN 1 ELSE 0 END) AS late_orders,
    ROUND(
        100.0 * SUM(CASE WHEN Delay > 0 THEN 1 ELSE 0 END) / COUNT(*),
        2
    ) AS delay_percentage
FROM supply_chain
GROUP BY "Shipping Mode"
ORDER BY delay_percentage DESC;

10.Strongest predictors of late delivery
SELECT
    "Shipping Mode",

    COUNT(*) AS total_orders,

    ROUND(
        AVG(
            CASE
                WHEN Delay > 0 THEN 1.0
                ELSE 0.0
            END
        ) * 100,
        2
    ) AS delay_percentage

FROM supply_chain

GROUP BY "Shipping Mode"

ORDER BY delay_percentage DESC;
